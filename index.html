    <script>

        let noClicks = 0;

        const toasts = [

            "⚠️ Error: Selection not recognized. Please try harder.",

            "❌ 404: 'Hard Pass' button not found in your heart.",

            "📡 Signal weak! The server only accepts positive vibes.",

            "🛑 Okay, fine! But that button is literally running away from you now."

        ];


        function showToast(msg) {

            const t = document.getElementById('toast');

            t.innerText = msg;

            t.style.bottom = '40px';

            setTimeout(() => { t.style.bottom = '-100px'; }, 2500);

        }


        function moveButton(btnId) {

            const btn = document.getElementById(btnId);

            const x = Math.random() * (window.innerWidth - 150);

            const y = Math.random() * (window.innerHeight - 80);

            btn.style.position = 'fixed';

            btn.style.left = x + 'px';

            btn.style.top = y + 'px';

            btn.style.zIndex = '1000';

        }


        function handleNo(page) {

            noClicks++;

            const btnId = 'no' + page;

            

            if (noClicks <= 3) {

                showToast(toasts[noClicks - 1]);

            } else {

                showToast(toasts[3]);

                moveButton(btnId);

            }

        }


        function nextPage(num) {

            noClicks = 0; // Reset for the next page

            document.querySelectorAll('.container').forEach(el => el.classList.add('hidden'));

            document.getElementById('page' + num).classList.remove('hidden');

            

            if (num === 3) {

                const duration = 5 * 1000;

                const animationEnd = Date.now() + duration;

                const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 0 };


                const interval = setInterval(function() {

                    const timeLeft = animationEnd - Date.now();

                    if (timeLeft <= 0) return clearInterval(interval);

                    const particleCount = 50 * (timeLeft / duration);

                    confetti({ ...defaults, particleCount, origin: { x: Math.random(), y: Math.random() - 0.2 } });

                }, 250);

            }

        }
